cmake_minimum_required(VERSION 3.20)
project(ctr_engine LANGUAGES CXX)
include(CTest)
add_executable(ctr_probe src/probe.cpp)
target_compile_features(ctr_probe PRIVATE cxx_std_20)
set_target_properties(ctr_probe PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(ctr_probe PRIVATE /W4 /WX)
else()
  target_compile_options(ctr_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME native_protocol COMMAND ctr_probe)
  set_tests_properties(native_protocol PROPERTIES PASS_REGULAR_EXPRESSION "ctr-probe-v1:42" TIMEOUT 5)
  add_test(NAME python_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/probe_test.py $<TARGET_FILE:ctr_probe>)
  set_tests_properties(python_protocol PROPERTIES TIMEOUT 10)
endif()
